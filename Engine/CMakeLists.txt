cmake_minimum_required(VERSION 3.20)
project(WhisperPasteEngine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../vendor/whisper.cpp" whisper)
add_executable(whisper-worker worker.cpp)
target_include_directories(whisper-worker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/whisper.cpp/examples")
target_link_libraries(whisper-worker PRIVATE common whisper json_cpp)
